Each factory function follows this pattern:
```typescript
import type { XxxInitMessage } from '../../../src/shared/messageContracts';

export function createXxxPayload(
    overrides?: Partial<XxxInitMessage['payload']>
): XxxInitMessage {
    return {
        type: 'xxxInit',
        payload: {
            // minimal valid defaults
            ...overrides,
        },
    };
}
```
