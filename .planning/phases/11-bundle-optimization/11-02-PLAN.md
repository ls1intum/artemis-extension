```bash
const meta = require('./dist/meta-webview.json');

  .reduce((sum, [, v]) => sum + v.bytes, 0);
console.log('Total Shiki:', (totalShiki / 1024 / 1024).toFixed(2) + ' MB');
"
```

**Step 4: Profile top dependencies by size**
```bash
node -e "
const meta = require('./dist/meta-webview.json');
const byPkg = {};
for (const [file, info] of Object.entries(meta.inputs)) {
  const match = file.match(/node_modules\/(@[^/]+\/[^/]+|[^/]+)/);
  if (match) {
    byPkg[match[1]] = (byPkg[match[1]] || 0) + info.bytes;
  }
}
const sorted = Object.entries(byPkg)
  .sort((a, b) => b[1] - a[1])

console.log('Top dependencies by size:');
sorted.forEach(([pkg, size]) => console.log('  ' + size + '  ' + pkg));
"
```
