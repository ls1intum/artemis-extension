5. **afterEach():** `switchBackFromWebview(driver)` in try/catch.

6. **Helper function `assertNoAxeViolations(viewName: string, driver: WebDriver)`:**
   ```typescript
   async function assertNoAxeViolations(viewName: string, driver: WebDriver): Promise<void> {
       const results = await runAxeInCurrentFrame(driver);
       if (results.violations.length > 0) {
           const summary = results.violations.map(v =>
               `[${v.impact}] ${v.id}: ${v.description} (${v.nodes.length} nodes)`
           ).join('\n  ');
           await takeScreenshot(driver, `a11y-fail-${viewName}`);
           assert.strictEqual(
               results.violations.length, 0,
               `${viewName} has ${results.violations.length} axe violations:\n  ${summary}`
           );
       }
   }
   ```

7. **12 individual test blocks:**

   **View 1 — Login (no auth needed):**
   ```typescript
   it('Login view should have zero axe violations', async function () {
       this.timeout(30000);
       // Login view only available when NOT logged in
       // Skip if already logged in (hasCredentials and login succeeded)
       if (hasCredentials) { this.skip(); } // Can't show login when logged in
       await openArtemisView();
       await switchToWebviewFrame(driver);
       await waitForElement(driver, 'form', 10000);
       await assertNoAxeViolations('login', driver);
   });
   ```

   Wait — the Login view is only shown when not authenticated. If we logged in during `before()`, we can't see Login anymore. Two approaches:
   - Option A: Test Login first (before logging in), then log in for the rest
   - Option B: Skip Login axe test when credentials are available

   **Use Option A:** Restructure so Login is tested BEFORE the login flow. Use nested `describe` blocks:

   ```typescript
   describe('Accessibility Tests (WCAG 2.1 AA)', function () {
       let driver: WebDriver;
       let hasCredentials = false;
       let username: string;
       let password: string;

       before(async function () {
           this.timeout(15000);
           driver = VSBrowser.instance.driver;
           await VSBrowser.instance.waitForWorkbench();
           try {
               ({ username, password } = getCredentials());
               hasCredentials = true;
           } catch { /* no credentials — only Login view testable */ }
       });

       afterEach(async function () {
           try { await switchBackFromWebview(driver); } catch { /* ignore */ }
       });

       // --- Pre-login tests ---
       it('Login view should have zero axe violations', async function () {
           this.timeout(30000);
           await openArtemisView();
           await switchToWebviewFrame(driver);
           await waitForElement(driver, 'form', 10000);
           await assertNoAxeViolations('login', driver);
       });

       // --- Login and test authenticated views ---
       describe('Authenticated views', function () {
           before(async function () {
               this.timeout(30000);
               if (!hasCredentials) { this.skip(); }
               await openArtemisView();
               await switchToWebviewFrame(driver);
               const usernameInput = await waitForElement(driver, '#username');
               await usernameInput.clear();
               await usernameInput.sendKeys(username);
               const passwordInput = await waitForElement(driver, '#password');
               await passwordInput.clear();
               await passwordInput.sendKeys(password);
               const submitBtn = await waitForElement(driver, 'button[type="submit"]');
               await submitBtn.click();
               await switchBackFromWebview(driver);
               await driver.sleep(5000);
           });

           after(async function () {
               this.timeout(15000);
               const workbench = new Workbench();
               await workbench.executeCommand('Logout from Artemis');
               await driver.sleep(2000);
           });

           // Views 2-12 here...
       });
   });
   ```
