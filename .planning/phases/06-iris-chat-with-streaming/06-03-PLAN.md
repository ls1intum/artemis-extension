**4. Message listener (critical — matches legacy format):**

Set up `window.addEventListener('message', handler)` in useEffect to receive messages from ChatWebviewProvider. The extension sends messages in LEGACY format with `message.command` (NOT `message.type`). Handle:

```typescript
useEffect(() => {
    const handler = (event: MessageEvent) => {
        const message = event.data;
        switch (message.command) {
            case 'updateIrisState':
                store.setIrisState(message.state);
                break;
            case 'showContextPicker':
                store.setIrisState(message.state);
                // Open the context selector dropdown in picker mode
                setForceContextPicker(true);
                break;
            case 'addMessage':
                if (message.message) {
                    store.addMessage({
                        id: message.message.id,
                        localId: crypto.randomUUID(),
                        role: message.message.role,
                        content: message.message.content,
                        timestamp: message.message.timestamp,
                        helpful: message.message.helpful,
                        status: 'sent',
                    });
                }
                break;
            case 'loadMessages':
                if (message.messages) {
                    store.setMessages(message.messages.map(m => ({
                        id: m.id,
                        localId: crypto.randomUUID(),
                        role: m.role,
                        content: m.content,
                        timestamp: m.timestamp,
                        helpful: m.helpful,
                        status: 'sent',
                    })));
                }
                break;
            case 'clearChatMessages':
                store.clearMessages();
                break;
            case 'updateReferencedFiles':
                store.setReferencedFiles({
                    includedFiles: message.includedFiles || [],
                    excludedFiles: message.excludedFiles || [],
                    totalCount: message.totalCount || 0,
                });
                break;
            case 'updateWebSocketStatus':
                store.setWebSocketConnected(message.isConnected);
                break;
            case 'showDisabledState':
                store.setDisabledMessage(message.message);
                break;
            case 'hideDisabledState':
                store.setDisabledMessage(null);
                break;
            case 'updateNoAiStatus':
                store.setNoAiDetected(message.isNoAiDetected);
                break;
        }
    };
    window.addEventListener('message', handler);
    return () => window.removeEventListener('message', handler);
}, []);
```

**e. Handle the ready signal:**

The React webview sends `{ type: 'ready' }` on mount (see `index.tsx` line 31). The ChatWebviewProvider currently handles `chatViewReady` (legacy command). We need to handle BOTH:

In `_handleMessage()`, add a case for the React ready signal:
```typescript
case 'ready':  // React ready signal (type: 'ready' comes through as command: undefined, type: 'ready')
    // The React index.tsx sends { type: 'ready' } not { command: 'ready' }
    // Handle this in the message listener
    break;
```

Actually, looking more carefully: the message listener in `resolveWebviewView` uses `webviewView.webview.onDidReceiveMessage(message => this._handleMessage(message))`. The `_handleMessage` switches on `message.command`. But the React ready signal sends `{ type: 'ready' }`, which has no `command` field.

**Fix:** Update `_handleMessage` to check for both:
```typescript
private _handleMessage(message: any): void {
    // Handle React ready signal (sent as { type: 'ready' })
    if (message.type === 'ready') {
        this._postSnapshot();
        this._postNoAiStatus(this._noAiDetectionService.isNoAiEnabled);
        return;
    }

    switch (message.command) {
        // ... existing cases (keep all of them)
        case 'chatViewReady':  // Legacy ready signal — keep for backward compatibility during transition
            this._postSnapshot();
            this._postNoAiStatus(this._noAiDetectionService.isNoAiEnabled);
            break;
        // ... rest of cases
    }
}
```
